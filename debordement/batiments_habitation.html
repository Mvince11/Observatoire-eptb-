    function calculerStatsIndicateur(champ, classes) {

      }

        totalCommunes: indicateursGeojson.features.length,
        communesExposees,
        totalBatiments,
        parClasse

      // Analyse "AB 12 Poitiers" ou "Poitiers AB 12"
      function analyserParcelle(q) {
        q = q.trim();
        let m = q.match(/^([A-Za-z]{1,2})\s*0*(\d{1,4})\s+(.{2,})$/);     // section numéro commune
        if (m) return { section: m[1], numero: m[2], commune: m[3] };
        m = q.match(/^(.{2,}?)\s+([A-Za-z]{1,2})\s*0*(\d{1,4})$/);        // commune section numéro
        if (m) return { section: m[2], numero: m[3], commune: m[1] };
        return null;
      }
      
      async function chercherParcelles(query, signal) {
        const a = analyserParcelle(query);
        if (!a) return [];
      
        const section = a.section.toUpperCase().padStart(2, '0');
        const numero  = a.numero.padStart(4, '0');
      
        // 1. Commune(s) correspondante(s)
        const rc = await fetch(
          `https://geo.api.gouv.fr/communes?nom=${encodeURIComponent(a.commune)}` +
          `&fields=nom,code,centre&boost=population&limit=3`, { signal });
        const communes = (await rc.json()).filter(c => c.centre && dansPerimetre(c.centre.coordinates));
      
        // 2. Parcelle dans chaque commune
        const resultats = await Promise.all(communes.map(async c => {
          const url = `https://apicarto.ign.fr/api/cadastre/parcelle?code_insee=${c.code}` +
                      `&section=${section}&numero=${numero}`;
          const r = await fetch(url, { signal });
          if (!r.ok) return [];
          const gj = await r.json();
          return (gj.features || []).map(f => {
            const bb = bboxGeom(f.geometry);
            const center = centreGeom(bb);
            const nom = `Parcelle ${section} ${numero} – ${c.nom}`;
            return {
              type: 'Feature',
              geometry: { type: 'Point', coordinates: center },
              place_name: nom,
              text: nom,
              properties: { ...f.properties, _insee: c.code },
              place_type: ['parcel'],
              center,
              bbox: bb
            };
          });
        }));
        return resultats.flat();
      }

        function formatHabitation(v) {
          if (v === null || v === undefined) return 'Non concerné';
          const s = String(v).trim().toLowerCase();
          if (['', 'na', 'nan', 'null', 'undefined'].includes(s)) return 'Non concerné';
          const n = Number(s.replace(',', '.'));
          if (isNaN(n)) return '1 à 19';
          if (n === 0)  return 'Aucun';
          if (n < 0)    return '1 à 19';
          return n.toLocaleString('fr-FR');
        }

        function chargerAleaMode(mode) {
          const etat = etatAleaBatiments[mode];
          if (etat.gj) return Promise.resolve(etat.gj);
          if (etat.promesse) return etat.promesse;
          etat.promesse = fetch(CONFIG_MODES[mode].batimentsAleaUrl)
            .then(r => {
              if (!r.ok) throw new Error('Couche des bâtiments en zone inondable introuvable');
              return r.json();
            })
            .then(gj => {
              etat.gj = gj;
              etat.ids = [...new Set(
                (gj.features || []).map(f => f.properties?.id_bat).filter(Boolean)
              )];
              return gj;
            });
          return etat.promesse;
        }

        function contourEnSurface(data) {
          if (!data || !data.features) return data;
          return {
            ...data,
            features: data.features.map(ft => {
              const g = ft.geometry;
              if (!g) return ft;
              if (g.type === 'LineString') {
                return { ...ft, geometry: { type: 'Polygon', coordinates: [g.coordinates] } };
              }
              if (g.type === 'MultiLineString') {
                return { ...ft, geometry: { type: 'Polygon', coordinates: g.coordinates } };
              }
              return ft;
            })
          };
        }

          features.forEach(f => walk(f.geometry.coordinates));
          return b;

Intitulé

Nombre de bâtiments d’habitation en zone inondable

Description

L’évaluation du nombre de bâtiments d’habitation permet de rendre compte des dommages causés par une inondation.

Modalité de calcul

L’indicateur consiste à compter les bâtiments d’habitation qui sont dans la zone d’inondation. Le résultat est agrégé à la commune.

Point de vigilance

La nature des bâtiments est issue de la sémantique présente dans la base des bâtiments de la BDTOPO. Cependant, il y a un croisement effectué avec les fichiers fonciers du CEREMA pour déterminer si le bâtiment contient des logements. Ceci peut expliquer que certains bâtiments autres que « Résidentiel » sont reconnus comme logements.
