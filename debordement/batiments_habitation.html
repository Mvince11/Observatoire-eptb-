      function analyserParcelle(q) {
        q = q.trim();
        let m = q.match(/^([A-Za-z]{1,2})\s*0*(\d{1,4})\s+(.{2,})$/);     // section numéro commune
        if (m) return { section: m[1], numero: m[2], commune: m[3] };
        m = q.match(/^(.{2,}?)\s+([A-Za-z]{1,2})\s*0*(\d{1,4})$/);        // commune section numéro
        if (m) return { section: m[2], numero: m[3], commune: m[1] };
        return null;
      }

        function formatHabitation(v) {
          if (v === null || v === undefined) return 'Non concerné';
          const s = String(v).trim().toLowerCase();
          if (['', 'na', 'nan', 'null', 'undefined'].includes(s)) return 'Non concerné';
          const n = Number(s.replace(',', '.'));
          if (isNaN(n)) return '1 à 19';
          if (n === 0)  return 'Aucun';
          if (n < 0)    return '1 à 19';
          return n.toLocaleString('fr-FR');

        function creerCoucheVectorielle(layer, beforeId) {
          if (map.getLayer(layer.id)) return;

          if (layer.pmtiles) {
            ensurePmtilesSource(layer.source, layer.url);
            const layerConfig = {
              id: layer.id,
              type: layer.type,
              source: layer.source,
              'source-layer': layer.sourceLayer,
              paint: {},
              layout: { ...(layer.layout || {}) }
            };
            if (layer.type === 'fill') {
              layerConfig.paint['fill-color'] = layer.color || '#0284c7';
              layerConfig.paint['fill-opacity'] = layer.opacity ?? 0.4;
              layerConfig.paint['fill-outline-color'] = layer.outlineColor || layer.color || '#0369a1';
            } else if (layer.type === 'line') {
              layerConfig.paint['line-color'] = layer.color || '#0369a1';
              layerConfig.paint['line-opacity'] = layer.opacity ?? 0.85;
              layerConfig.paint['line-width'] = layer.width || 1.5;

                layerConfig.layout['line-cap'] = 'round';
                layerConfig.layout['line-join'] = 'round';
              }
            } else if (layer.type === 'fill-extrusion') {
              layerConfig.paint['fill-extrusion-color'] = layer.colorByType === 'ff_dteloctxt'
                ? COULEURS_BATIMENTS
                : (layer.color || '#94a3b8');
              layerConfig.paint['fill-extrusion-height'] = ['coalesce', ['get', layer.heightProperty || 'hauteur'], 5];
              layerConfig.paint['fill-extrusion-base'] = 0;
              layerConfig.paint['fill-extrusion-opacity'] = layer.opacity ?? 0.88;

              type: 'geojson',
              data: layer.data

            id: layer.id,

          features.forEach(f => walk(f.geometry.coordinates));
          return b;
